<script>
    const AFF_ID = "17365550406";

    function process() {
        const url = document.getElementById('rawLink').value.trim();
        if(!url) return alert("Vui lòng dán link!");

        // Xử lý ghép link thủ công
        const encoded = encodeURIComponent(url);
        const result = `https://s.shopee.vn/an_redir?origin_link=${encoded}&affiliate_id=${AFF_ID}`;

        document.getElementById('finalLink').innerText = result;
        document.getElementById('resultArea').style.display = 'block';
    }

    function copy() {
        const text = document.getElementById('finalLink').innerText;
        navigator.clipboard.writeText(text);
        alert("Đã copy thành công!");
    }
</script>
